package main

import (
	"bytes"
	"fmt"
	"os"
	"os/exec"
	"path/filepath"
)

func main() {
	if err := check(); err != nil {
		fmt.Fprintf(os.Stderr, "ledgerroll end-to-end check failed: %v\n", err)
		os.Exit(1)
	}
}

func check() error {
	repository, err := os.Getwd()
	if err != nil {
		return err
	}
	temporary, err := os.MkdirTemp("", "ledgerroll-e2e-")
	if err != nil {
		return err
	}
	defer os.RemoveAll(temporary)

	binary := filepath.Join(temporary, "ledgerroll")
	build := exec.Command("go", "build", "-o", binary, ".")
	build.Dir = repository
	if output, err := build.CombinedOutput(); err != nil {
		return fmt.Errorf("build command failed: %w\n%s", err, output)
	}

	ledger := "" +
		`{"id":"c2","at":"2024-01-01T11:00:00+01:00","account":"bob","currency":"EUR","kind":"charge","amount":"5.00"}` + "\n" +
		`{"id":"r1","at":"2024-01-01T10:30:00Z","account":"alice","currency":"USD","kind":"refund","amount":"2.25","ref":"c1"}` + "\n" +
		`{"id":"c1","at":"2024-01-01T09:00:00Z","account":"alice","currency":"USD","kind":"charge","amount":"10.00"}` + "\n" +
		`{"id":"c3","at":"2024-01-01T11:00:00Z","account":"alice","currency":"USD","kind":"charge","amount":"1.20"}` + "\n" +
		`{"id":"r2","at":"2024-01-01T12:00:00Z","account":"bob","currency":"EUR","kind":"refund","amount":"1.00","ref":"c2"}` + "\n"

	tests := []testCase{
		{
			name:      "ordered relationships and grouping",
			arguments: []string{"summarize"},
			input:     ledger,
			stdout:    "{\"from\":null,\"to\":null,\"transactions\":5,\"totals\":[{\"account\":\"alice\",\"currency\":\"USD\",\"amount\":\"8.95\"},{\"account\":\"bob\",\"currency\":\"EUR\",\"amount\":\"4.00\"}]}\n",
		},
		{
			name:      "normalized half-open window",
			arguments: []string{"summarize", "--to=2024-01-01T12:00:00Z", "--from", "2024-01-01T11:00:00+01:00"},
			input:     ledger,
			stdout:    "{\"from\":\"2024-01-01T10:00:00Z\",\"to\":\"2024-01-01T12:00:00Z\",\"transactions\":3,\"totals\":[{\"account\":\"alice\",\"currency\":\"USD\",\"amount\":\"-1.05\"},{\"account\":\"bob\",\"currency\":\"EUR\",\"amount\":\"5.00\"}]}\n",
		},
		{
			name:      "arbitrary precision and blank lines",
			arguments: []string{"summarize"},
			input: "  \n" +
				`{"id":"big","at":"2024-02-01T00:00:00Z","account":"z","currency":"USD","kind":"charge","amount":"9007199254740993.01"}` + "\n\n" +
				`{"id":"small","at":"2024-02-01T00:00:01Z","account":"z","currency":"USD","kind":"refund","amount":"0.01","ref":"big"}` + "\n",
			stdout: "{\"from\":null,\"to\":null,\"transactions\":2,\"totals\":[{\"account\":\"z\",\"currency\":\"USD\",\"amount\":\"9007199254740993.00\"}]}\n",
		},
		{
			name:      "empty input",
			arguments: []string{"summarize"},
			input:     "\n \t\n",
			stdout:    "{\"from\":null,\"to\":null,\"transactions\":0,\"totals\":[]}\n",
		},
		{
			name:      "duplicate id",
			arguments: []string{"summarize"},
			input: `{"id":"x","at":"2024-01-01T00:00:00Z","account":"a","currency":"USD","kind":"charge","amount":"1.00"}` + "\n" +
				`{"id":"x","at":"2024-01-02T00:00:00Z","account":"a","currency":"USD","kind":"charge","amount":"2.00"}` + "\n",
			status: 2,
			stderr: "invalid record at line 2: duplicate id \"x\"\n",
		},
		{
			name:      "invalid amount",
			arguments: []string{"summarize"},
			input:     `{"id":"x","at":"2024-01-01T00:00:00Z","account":"a","currency":"USD","kind":"charge","amount":"01.00"}` + "\n",
			status:    2,
			stderr:    "invalid record at line 1: invalid amount \"01.00\"\n",
		},
		{
			name:      "stable timestamp order",
			arguments: []string{"summarize"},
			input: `{"id":"r","at":"2024-01-01T00:00:00Z","account":"a","currency":"USD","kind":"refund","amount":"1.00","ref":"c"}` + "\n" +
				`{"id":"c","at":"2023-12-31T19:00:00-05:00","account":"a","currency":"USD","kind":"charge","amount":"1.00"}` + "\n",
			status: 2,
			stderr: "invalid record at line 1: refund ref \"c\" not found\n",
		},
		{
			name:      "refund mismatch",
			arguments: []string{"summarize"},
			input: `{"id":"c","at":"2024-01-01T00:00:00Z","account":"a","currency":"USD","kind":"charge","amount":"2.00"}` + "\n" +
				`{"id":"r","at":"2024-01-01T00:00:01Z","account":"a","currency":"EUR","kind":"refund","amount":"1.00","ref":"c"}` + "\n",
			status: 2,
			stderr: "invalid record at line 2: refund \"r\" does not match charge \"c\"\n",
		},
		{
			name:      "cumulative refund limit outside window",
			arguments: []string{"summarize", "--to", "2024-01-02T00:00:00Z"},
			input: `{"id":"c","at":"2024-01-01T00:00:00Z","account":"a","currency":"USD","kind":"charge","amount":"2.00"}` + "\n" +
				`{"id":"r1","at":"2024-01-03T00:00:00Z","account":"a","currency":"USD","kind":"refund","amount":"1.50","ref":"c"}` + "\n" +
				`{"id":"r2","at":"2024-01-04T00:00:00Z","account":"a","currency":"USD","kind":"refund","amount":"0.51","ref":"c"}` + "\n",
			status: 2,
			stderr: "invalid record at line 3: refunds for \"c\" exceed charge\n",
		},
		{
			name:      "trailing JSON",
			arguments: []string{"summarize"},
			input:     `{"id":"x"} {"id":"y"}` + "\n",
			status:    2,
			stderr:    "invalid record at line 1: invalid JSON\n",
		},
		{
			name:      "invalid from",
			arguments: []string{"summarize", "--from", "yesterday"},
			status:    2,
			stderr:    "invalid --from: yesterday\n",
		},
		{
			name:      "invalid range",
			arguments: []string{"summarize", "--from=2024-01-01T00:00:00Z", "--to=2024-01-01T00:00:00Z"},
			status:    2,
			stderr:    "invalid time range\n",
		},
		{
			name:      "usage",
			arguments: []string{"summarize", "--from", "2024-01-01T00:00:00Z", "extra"},
			status:    2,
			stderr:    "usage: ledgerroll summarize [--from TIME] [--to TIME]\n",
		},
	}

	for _, test := range tests {
		if err := test.run(binary, temporary); err != nil {
			return err
		}
	}
	return nil
}

type testCase struct {
	name      string
	arguments []string
	input     string
	status    int
	stdout    string
	stderr    string
}

func (test testCase) run(binary, workingDirectory string) error {
	command := exec.Command(binary, test.arguments...)
	command.Dir = workingDirectory
	command.Stdin = bytes.NewBufferString(test.input)
	var stdout, stderr bytes.Buffer
	command.Stdout = &stdout
	command.Stderr = &stderr
	err := command.Run()
	gotStatus := 0
	if err != nil {
		exitError, ok := err.(*exec.ExitError)
		if !ok {
			return fmt.Errorf("%s could not run: %w", test.name, err)
		}
		gotStatus = exitError.ExitCode()
	}
	if gotStatus != test.status {
		return fmt.Errorf("%s exit status = %d, want %d", test.name, gotStatus, test.status)
	}
	if stdout.String() != test.stdout {
		return fmt.Errorf("%s stdout = %q, want %q", test.name, stdout.String(), test.stdout)
	}
	if stderr.String() != test.stderr {
		return fmt.Errorf("%s stderr = %q, want %q", test.name, stderr.String(), test.stderr)
	}
	return nil
}
