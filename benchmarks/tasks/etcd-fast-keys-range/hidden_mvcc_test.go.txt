package mvcc

import (
	"reflect"
	"testing"

	"go.uber.org/zap/zaptest"

	"go.etcd.io/etcd/server/v3/lease"
	betesting "go.etcd.io/etcd/server/v3/storage/backend/testing"
)

func TestHPatchBenchmarkFastKeysRangeLimitAndCount(t *testing.T) {
	backend, _ := betesting.NewDefaultTmpBackend(t)
	store := NewStore(zaptest.NewLogger(t), backend, &lease.FakeLessor{}, StoreConfig{})
	defer cleanup(store, backend)

	store.Put([]byte("a"), []byte("value-a"), lease.NoLease)
	store.Put([]byte("b"), []byte("value-b"), lease.NoLease)
	store.Put([]byte("c"), []byte("value-c"), lease.NoLease)
	store.DeleteRange([]byte("b"), nil)

	tests := []struct {
		name           string
		limit          int64
		withTotalCount bool
		wantKeys       []string
		wantCount      int
	}{
		{
			name:      "limited scan stops without total count",
			limit:     1,
			wantKeys:  []string{"a"},
			wantCount: 1,
		},
		{
			name:           "limited scan retains full count",
			limit:          1,
			withTotalCount: true,
			wantKeys:       []string{"a"},
			wantCount:      2,
		},
		{
			name:      "unlimited scan",
			wantKeys:  []string{"a", "c"},
			wantCount: 2,
		},
	}

	for _, test := range tests {
		t.Run(test.name, func(t *testing.T) {
			options := RangeOptions{
				Limit:          test.limit,
				WithTotalCount: test.withTotalCount,
			}
			fast := reflect.ValueOf(&options).Elem().FieldByName("FastKeysOnly")
			if !fast.IsValid() {
				t.Fatal("FastKeysOnly option is missing")
			}
			fast.SetBool(true)

			result, err := store.Range(t.Context(), []byte("a"), []byte("z"), options)
			if err != nil {
				t.Fatalf("Range returned error: %v", err)
			}
			if result.Count != test.wantCount {
				t.Errorf("Count = %d, want %d", result.Count, test.wantCount)
			}
			if len(result.KVs) != len(test.wantKeys) {
				t.Fatalf("returned %d keys, want %d", len(result.KVs), len(test.wantKeys))
			}
			for i, wantKey := range test.wantKeys {
				kv := result.KVs[i]
				if string(kv.Key) != wantKey {
					t.Errorf("key[%d] = %q, want %q", i, kv.Key, wantKey)
				}
				if len(kv.Value) != 0 {
					t.Errorf("value[%d] = %q, want no backend value", i, kv.Value)
				}
				if kv.CreateRevision == 0 || kv.ModRevision == 0 || kv.Version != 1 {
					t.Errorf("metadata[%d] = create:%d mod:%d version:%d", i, kv.CreateRevision, kv.ModRevision, kv.Version)
				}
			}
		})
	}
}
