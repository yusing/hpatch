package txn

import (
	"context"
	"reflect"
	"testing"

	"go.uber.org/zap/zaptest"

	pb "go.etcd.io/etcd/api/v3/etcdserverpb"
	"go.etcd.io/etcd/api/v3/mvccpb"
	"go.etcd.io/etcd/pkg/v3/traceutil"
	"go.etcd.io/etcd/server/v3/storage/mvcc"
)

type hpatchBenchmarkTxnRead struct {
	options mvcc.RangeOptions
}

func (r *hpatchBenchmarkTxnRead) FirstRev() int64 { return 1 }
func (r *hpatchBenchmarkTxnRead) Rev() int64      { return 3 }
func (r *hpatchBenchmarkTxnRead) End()            {}

func (r *hpatchBenchmarkTxnRead) Range(context.Context, []byte, []byte, mvcc.RangeOptions) (*mvcc.RangeResult, error) {
	panic("unreachable")
}

func (r *hpatchBenchmarkTxnRead) captureRange(_ context.Context, _ []byte, _ []byte, options mvcc.RangeOptions) (*mvcc.RangeResult, error) {
	r.options = options
	return &mvcc.RangeResult{
		KVs: []*mvccpb.KeyValue{{
			Key:            []byte("key"),
			Value:          []byte("value"),
			CreateRevision: 2,
			ModRevision:    3,
			Version:        1,
		}},
		Count: 1,
		Rev:   3,
	}, nil
}

type hpatchBenchmarkReadAdapter struct {
	*hpatchBenchmarkTxnRead
}

func (r hpatchBenchmarkReadAdapter) Range(ctx context.Context, key, end []byte, options mvcc.RangeOptions) (*mvcc.RangeResult, error) {
	return r.captureRange(ctx, key, end, options)
}

func TestHPatchBenchmarkKeysOnlyRouting(t *testing.T) {
	tests := []struct {
		name       string
		request    *pb.RangeRequest
		wantFast   bool
		totalCount bool
	}{
		{
			name: "keys only uses fast path",
			request: &pb.RangeRequest{
				Key:      []byte("a"),
				RangeEnd: []byte("z"),
				KeysOnly: true,
			},
			wantFast: true,
		},
		{
			name: "value sort requires backend values",
			request: &pb.RangeRequest{
				Key:        []byte("a"),
				RangeEnd:   []byte("z"),
				KeysOnly:   true,
				SortTarget: pb.RangeRequest_VALUE,
			},
		},
		{
			name: "ordinary range requires backend values",
			request: &pb.RangeRequest{
				Key:      []byte("a"),
				RangeEnd: []byte("z"),
			},
			totalCount: true,
		},
	}

	for _, test := range tests {
		t.Run(test.name, func(t *testing.T) {
			read := &hpatchBenchmarkTxnRead{}
			ctx, _ := traceutil.EnsureTrace(t.Context(), zaptest.NewLogger(t), "hpatch benchmark")
			_, err := executeRange(ctx, zaptest.NewLogger(t), hpatchBenchmarkReadAdapter{read}, test.request, test.totalCount)
			if err != nil {
				t.Fatalf("executeRange returned error: %v", err)
			}

			options := reflect.ValueOf(read.options)
			fast := options.FieldByName("FastKeysOnly")
			if !fast.IsValid() {
				t.Fatal("FastKeysOnly option is missing")
			}
			if got := fast.Bool(); got != test.wantFast {
				t.Errorf("FastKeysOnly = %t, want %t", got, test.wantFast)
			}
			if read.options.WithTotalCount != test.totalCount {
				t.Errorf("WithTotalCount = %t, want %t", read.options.WithTotalCount, test.totalCount)
			}
		})
	}
}
