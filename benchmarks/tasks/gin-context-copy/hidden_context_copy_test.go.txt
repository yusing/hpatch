package gin

import (
	"errors"
	"net/http"
	"net/http/httptest"
	"testing"
)

func TestHPatchBenchmarkContextCopyIsolation(t *testing.T) {
	c, _ := CreateTestContext(httptest.NewRecorder())
	c.Request, _ = http.NewRequest(http.MethodGet, "/", nil)
	_ = c.Error(errors.New("first"))
	_ = c.Error(errors.New("second"))
	c.SetAccepted("application/json", "text/html")

	cp := c.Copy()
	if len(cp.Errors) != 2 || len(cp.Accepted) != 2 {
		t.Fatalf("Context.Copy omitted request state: errors=%d accepted=%d", len(cp.Errors), len(cp.Accepted))
	}
	_ = cp.Error(errors.New("third"))
	cp.Accepted[0] = "text/plain"
	if len(c.Errors) != 2 || c.Accepted[0] != "application/json" {
		t.Fatalf("Context.Copy shared mutable request state: errors=%d accepted=%q", len(c.Errors), c.Accepted)
	}

	empty, _ := CreateTestContext(httptest.NewRecorder())
	empty.Request, _ = http.NewRequest(http.MethodGet, "/", nil)
	emptyCopy := empty.Copy()
	if emptyCopy.Errors != nil || emptyCopy.Accepted != nil {
		t.Fatalf("Context.Copy changed nil request state: errors=%v accepted=%v", emptyCopy.Errors, emptyCopy.Accepted)
	}
}
