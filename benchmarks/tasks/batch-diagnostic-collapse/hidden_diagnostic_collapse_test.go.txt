package diagnosticcollapse

import (
	"slices"
	"strings"
	"testing"
)

func TestHPatchBenchmarkCollapse(t *testing.T) {
	tests := []struct {
		name  string
		input []string
		want  []Summary
	}{
		{
			name:  "consecutive runs preserve order",
			input: []string{"alpha", "alpha", "beta", "alpha", "alpha", "alpha"},
			want: []Summary{
				{Text: "alpha", Count: 2},
				{Text: "beta", Count: 1},
				{Text: "alpha", Count: 3},
			},
		},
		{
			name:  "empty and Unicode diagnostics remain distinct",
			input: []string{"", "", "故障", "故障", ""},
			want: []Summary{
				{Text: "", Count: 2},
				{Text: "故障", Count: 2},
				{Text: "", Count: 1},
			},
		},
		{
			name:  "long diagnostic",
			input: []string{strings.Repeat("x", 128_000), strings.Repeat("x", 128_000)},
			want:  []Summary{{Text: strings.Repeat("x", 128_000), Count: 2}},
		},
		{
			name: "no diagnostics",
		},
	}

	for _, test := range tests {
		t.Run(test.name, func(t *testing.T) {
			original := slices.Clone(test.input)
			got := Collapse(test.input)
			if !slices.Equal(test.input, original) {
				t.Fatalf("Collapse mutated input: got %q, want %q", test.input, original)
			}
			if !slices.Equal(got, test.want) {
				t.Fatalf("Collapse() = %#v, want %#v", got, test.want)
			}
		})
	}
}
