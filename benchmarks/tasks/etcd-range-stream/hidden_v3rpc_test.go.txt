package v3rpc

import (
	"testing"

	"google.golang.org/grpc/codes"
	"google.golang.org/grpc/status"

	pb "go.etcd.io/etcd/api/v3/etcdserverpb"
	"go.etcd.io/etcd/client/pkg/v3/types"
)

type hpatchBenchmarkRaftStatus struct{}

func (hpatchBenchmarkRaftStatus) MemberID() types.ID       { return 1 }
func (hpatchBenchmarkRaftStatus) Leader() types.ID         { return 2 }
func (hpatchBenchmarkRaftStatus) CommittedIndex() uint64    { return 3 }
func (hpatchBenchmarkRaftStatus) AppliedIndex() uint64      { return 4 }
func (hpatchBenchmarkRaftStatus) Term() uint64              { return 9 }

func TestHPatchBenchmarkRangeStreamValidation(t *testing.T) {
	tests := []struct {
		name string
		req  *pb.RangeRequest
		want codes.Code
	}{
		{
			name: "default ordering is accepted",
			req:  &pb.RangeRequest{Key: []byte("a")},
			want: codes.OK,
		},
		{
			name: "value sorting is rejected",
			req: &pb.RangeRequest{
				Key:        []byte("a"),
				SortTarget: pb.RangeRequest_VALUE,
				SortOrder:  pb.RangeRequest_ASCEND,
			},
			want: codes.Unimplemented,
		},
		{
			name: "revision filters are rejected",
			req:  &pb.RangeRequest{Key: []byte("a"), MinModRevision: 1},
			want: codes.Unimplemented,
		},
	}
	for _, test := range tests {
		t.Run(test.name, func(t *testing.T) {
			if got := status.Code(checkRangeStreamRequest(test.req)); got != test.want {
				t.Fatalf("checkRangeStreamRequest() code = %s, want %s", got, test.want)
			}
		})
	}
}

func TestHPatchBenchmarkStreamingHeaderPreservesRevision(t *testing.T) {
	header := header{
		clusterID: 7,
		memberID:  11,
		sg:        hpatchBenchmarkRaftStatus{},
	}
	response := &pb.ResponseHeader{Revision: 42}
	header.fillWithoutRevision(response)
	if response.ClusterId != 7 || response.MemberId != 11 || response.RaftTerm != 9 {
		t.Fatalf("header fields = cluster:%d member:%d term:%d", response.ClusterId, response.MemberId, response.RaftTerm)
	}
	if response.Revision != 42 {
		t.Fatalf("fillWithoutRevision changed pinned revision to %d", response.Revision)
	}
}
