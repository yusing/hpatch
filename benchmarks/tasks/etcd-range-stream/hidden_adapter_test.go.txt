package adapter

import (
	"context"
	"testing"

	"google.golang.org/grpc/codes"
	"google.golang.org/grpc/status"

	pb "go.etcd.io/etcd/api/v3/etcdserverpb"
)

func TestHPatchBenchmarkProxyAdapterRejectsRangeStream(t *testing.T) {
	_, err := (&kvs2kvc{}).RangeStream(context.Background(), &pb.RangeRequest{})
	if status.Code(err) != codes.Unimplemented {
		t.Fatalf("RangeStream error code = %s, want %s", status.Code(err), codes.Unimplemented)
	}
}
