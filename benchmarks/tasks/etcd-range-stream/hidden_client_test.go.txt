package clientv3

import (
	"context"
	"testing"

	"google.golang.org/grpc"
	"google.golang.org/grpc/codes"
	"google.golang.org/grpc/status"

	pb "go.etcd.io/etcd/api/v3/etcdserverpb"
)

type hpatchBenchmarkKVClient struct {
	pb.KVClient
	calls int
}

func (c *hpatchBenchmarkKVClient) RangeStream(context.Context, *pb.RangeRequest, ...grpc.CallOption) (pb.KV_RangeStreamClient, error) {
	c.calls++
	return nil, status.Error(codes.Unavailable, "stream unavailable")
}

func TestHPatchBenchmarkRetryForwardsRangeStream(t *testing.T) {
	fake := &hpatchBenchmarkKVClient{}
	client := &retryKVClient{kc: fake}
	_, err := client.RangeStream(context.Background(), &pb.RangeRequest{})
	if status.Code(err) != codes.Unavailable {
		t.Fatalf("RangeStream error code = %s, want %s", status.Code(err), codes.Unavailable)
	}
	if fake.calls != 1 {
		t.Fatalf("RangeStream calls = %d, want 1", fake.calls)
	}
}
