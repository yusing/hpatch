package txn

import (
	"testing"

	"go.uber.org/zap/zaptest"

	pb "go.etcd.io/etcd/api/v3/etcdserverpb"
	"go.etcd.io/etcd/server/v3/lease"
	betesting "go.etcd.io/etcd/server/v3/storage/backend/testing"
	"go.etcd.io/etcd/server/v3/storage/mvcc"
)

func TestHPatchBenchmarkNormalRangeStillSupportsSortingAndRevisionFilters(t *testing.T) {
	backend, _ := betesting.NewDefaultTmpBackend(t)
	store := mvcc.New(zaptest.NewLogger(t), backend, &lease.FakeLessor{}, mvcc.StoreConfig{})
	t.Cleanup(func() { _ = store.Close() })
	store.Put([]byte("a"), []byte("2"), lease.NoLease)
	filterRevision := store.Put([]byte("b"), []byte("1"), lease.NoLease)

	response, _, err := Range(t.Context(), zaptest.NewLogger(t), store, &pb.RangeRequest{
		Key:            []byte("a"),
		RangeEnd:       []byte("c"),
		SortTarget:     pb.RangeRequest_VALUE,
		SortOrder:      pb.RangeRequest_ASCEND,
		MinModRevision: filterRevision,
	}, true)
	if err != nil {
		t.Fatalf("Range returned error: %v", err)
	}
	if len(response.Kvs) != 1 || string(response.Kvs[0].Key) != "b" {
		t.Fatalf("Range keys = %q, want [b]", response.Kvs)
	}
}
